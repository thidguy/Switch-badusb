REM Demo save: slot 0 -> TXT:NS2 (Direct mode)
REM Uses PRGEDIT/PRGGET$/SAVEV (plain SAVE is restricted in the demo)
DEFAULTDELAY 60
DELAY 6000
STRINGLN PRGEDIT 0
STRINGLN NS2TEXT$="":FOR NS2I=1 TO PRGSIZE(0):NS2TEXT$=NS2TEXT$+PRGGET$()+CHR$(10):NEXT
STRINGLN SAVEV "TXT:NS2",NS2TEXT$
STRINGLN PRINT "SAVED",LEN(NS2TEXT$),PRGSIZE(0)
