REM dprobe_v21 zero-string + PUT mechanism test -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1040
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PUT(O,V#)
STRINGLN COPY NSR[0],O,P$(V#)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,J,V#,E0#,E3#,OD$,KH#,OFF#,DGP#,S#,SH#,MM#,S$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN S$=COPY(NSR[0],4,4)
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2Z ==")
STRINGLN NSR[1]=CHR$(0)
STRINGLN FOR J=1 TO 18:NSR[1]=NSR[1]+NSR[1]:NEXT
STRINGLN X$=PLOG("ZL "+STR$(LEN(NSR[1])))
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN OFF#=(KH#-E0#)/2
STRINGLN D$=COPY(NSR[0],OFF#,8)
STRINGLN DGP#=U#(LEFT$(D$,4))
STRINGLN SH#=U#(MID$(D$,4,4))
STRINGLN X$=PLOG("LDP "+HX$(DGP#)+" "+HX$(SH#))
STRINGLN COPY NSR[0],4,S$
STRINGLN X$=PLOG("KT "+STR$(U#(COPY(NSR[0],4,4))==U#(S$)))
STRINGLN IF !PTR(DGP#,16) || SH#!=262144 THEN X$=PLOG("ZBAD"):GOTO @OUT
STRINGLN X$=PLOG("== S3T ==")
STRINGLN X$=AIM(DGP#,2048)
STRINGLN D$=COPY(NSR[1],0,2048)
STRINGLN X$=PLOG("ZERO "+STR$(D$==CHR$(0)*2048))
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("== S4P ==")
STRINGLN X$=AIM(DGP#,262144)
STRINGLN X$=PLOG("AIM OK")
STRINGLN X$=PUT(16432,1)
STRINGLN X$=PLOG("P16432 OK")
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("== S5D ==")
STRINGLN X$=PLOG("DUMPING OFFSETS...")
STRINGLN X$=AIM(DGP#,262144)
STRINGLN FOR J=0 TO 1024
STRINGLN D$=COPY(NSR[1],J,8)
STRINGLN V#=U#(LEFT$(D$,4))
STRINGLN IF V#!=0 THEN
STRINGLN X$=PLOG("NZ"+STR$(J)+" "+HX$(V#))
STRINGLN ENDIF
STRINGLN IF J MOD 128==0 THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN X$=PLOG("== S6S ==")
STRINGLN X$=PLOG("SAFETY CHECK: aiming and reading still work")
STRINGLN X$=AIM(E0#,16)
STRINGLN D$=COPY(NSR[1],0,16)
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("R "+STR$(D$==V$))
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN @OUT
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
