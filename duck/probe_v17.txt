REM dprobe_v17 allocator pool map -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1036
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,J,V#,E0#,E1#,E3#,OD$,KH#,OFF#,S$,M$,B$,TE$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E1#=U#(MID$(V$,4,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN TE$=COPY(NSR[0],4,4)
STRINGLN X$=PLOG("TY "+STR$(TYPEOF(NSR))+" "+STR$(TYPEOF(NSR[0])))
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2H ==")
STRINGLN X$=AIM(E0#-32,32)
STRINGLN D$=COPY(NSR[1],0,32)
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("HL "+STR$(LEN(D$)))
STRINGLN IF LEN(D$)==32 THEN
STRINGLN FOR J=0 TO 24 STEP 8
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN X$=PLOG("AH"+STR$(J)+" "+HX$(V#)+" "+HX$(U#(MID$(D$,J+4,4))))
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN X$=PLOG("== S2P ==")
STRINGLN M$="MIDTESTMIDTESTMIDTESTMIDTESTMIDTESTMIDTESTMIDTESTMIDTESTMIDTESTMIDTEST"
STRINGLN B$="QZ"
STRINGLN FOR I=1 TO 16
STRINGLN B$=B$+B$
STRINGLN NEXT
STRINGLN X$=PLOG("BL "+STR$(LEN(B$)))
STRINGLN NSR[1]="AB"
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN OFF#=(KH#-E0#)/2
STRINGLN IF OFF#>=0 && PTR(KH#,16) THEN
STRINGLN D$=COPY(NSR[0],OFF#,8)
STRINGLN X$=PLOG("PS "+HX$(KH#)+" "+HX$(U#(LEFT$(D$,4)))+" "+HX$(U#(MID$(D$,4,4))))
STRINGLN ELSE
STRINGLN X$=PLOG("PSX "+HX$(KH#))
STRINGLN ENDIF
STRINGLN NSR[1]=M$
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN OFF#=(KH#-E0#)/2
STRINGLN IF OFF#>=0 && PTR(KH#,16) THEN
STRINGLN D$=COPY(NSR[0],OFF#,8)
STRINGLN X$=PLOG("PM "+HX$(KH#)+" "+HX$(U#(LEFT$(D$,4)))+" "+HX$(U#(MID$(D$,4,4))))
STRINGLN ELSE
STRINGLN X$=PLOG("PMX "+HX$(KH#))
STRINGLN ENDIF
STRINGLN NSR[1]=B$
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN OFF#=(KH#-E0#)/2
STRINGLN IF OFF#>=0 && PTR(KH#,16) THEN
STRINGLN D$=COPY(NSR[0],OFF#,8)
STRINGLN X$=PLOG("PL "+HX$(KH#)+" "+HX$(U#(LEFT$(D$,4)))+" "+HX$(U#(MID$(D$,4,4))))
STRINGLN ELSE
STRINGLN X$=PLOG("PLX "+HX$(KH#))
STRINGLN ENDIF
STRINGLN X$=PLOG("KBL "+STR$(LEN(NSR[1])))
STRINGLN COPY NSR[0],4,TE$
STRINGLN X$=PLOG("KT "+STR$(U#(COPY(NSR[0],4,4))==E1#))
STRINGLN X$=PLOG("== S3D ==")
STRINGLN D$=COPY(NSR[0],0,16)
STRINGLN X$=PLOG("DL "+STR$(LEN(D$))+" "+STR$(D$==V$))
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
