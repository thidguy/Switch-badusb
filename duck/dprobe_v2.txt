REM dprobe_v2 demo probe -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,B#,C#,T#,SH#,S#,M#,D#,G#,MM#,H$,QF$,GH$,DP$,CP$
STRINGLN VAR PHASE=0,FD=-1,POOL#=0,ZERO$=CHR$(0),I
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1015
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF W$(A#,N)
STRINGLN VAR S$
STRINGLN COPY NSR[2],0,P$(A#)+P$(N/2)
STRINGLN S$=COPY(NSR[0],0,N/2):COPY NSR[2],0,H$
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,Q$,J,AT,K#,V#,L#,KH#,N
STRINGLN PRINT "== S1 =="
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN PRINT "W",J,HX$(V#),PTR(V#,16)
STRINGLN NEXT
STRINGLN B#=U#(LEFT$(V$,4)):C#=U#(MID$(V$,8,4))+16:T#=U#(MID$(V$,12,4))+16
STRINGLN H$=P$(T#)+P$(4):COPY NSR[2],0,H$
STRINGLN PRINT "COR",LEN(NSR[0])==B#,NSR[0]==NSR[3],NSR[3]
STRINGLN IF !PTR(B#,16) || !PTR(T#,16) THEN PRINT "PTRBAD":RETURN 0
STRINGLN PRINT "== S2 =="
STRINGLN COPY NSR[0],1,"9"
STRINGLN PRINT "WT",NSR[3]
STRINGLN COPY NSR[0],1,"2"
STRINGLN PRINT "== S3 =="
STRINGLN Q$=COPY(ZERO$,0,2048)
STRINGLN FOR AT=0 TO 2
STRINGLN NSR[1]=KEY(2):COPY NSR[2],0,P$(B#+8)+P$(4):KH#=U#(NSR[0]):COPY NSR[2],0,H$
STRINGLN PRINT "KH",HX$(KH#),PTR(KH#,16)
STRINGLN IF PTR(KH#,16) THEN
STRINGLN COPY NSR[2],0,P$(FLOOR(KH#/4096)*4096)+P$(2048)
STRINGLN COPY Q$,0,NSR[0]:COPY NSR[2],0,H$
STRINGLN N=0
STRINGLN FOR J=0 TO 2040 STEP 8
STRINGLN V#=U#(MID$(Q$,J,4)):L#=U#(MID$(Q$,J+4,4))
STRINGLN IF L#==5 && V#>4096 && V#<549755813888 && N<25 THEN
STRINGLN PRINT "C",HX$(V#):N=N+1
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN PRINT "== S4 =="
STRINGLN NSR[1]=CHR$(0):FOR J=1 TO 18:NSR[1]=NSR[1]+NSR[1]:NEXT
STRINGLN SH#=U#(W$(B#+8,8))
STRINGLN PRINT "SH",HX$(SH#),PTR(SH#,16)
STRINGLN IF !PTR(SH#,16) THEN PRINT "SHBAD":RETURN 0
STRINGLN Q$=W$(SH#,16):S#=U#(LEFT$(Q$,4))
STRINGLN PRINT "S",HX$(S#),PTR(S#,16),"SL",HX$(U#(MID$(Q$,4,4)))
STRINGLN IF !PTR(S#,16) THEN PRINT "SBAD":RETURN 0
STRINGLN H$=P$(S#)+P$(4):COPY NSR[2],0,H$
STRINGLN PRINT "ZT",NSR[0]==LEFT$(ZERO$,4)
STRINGLN PRINT "DONE"
STRINGLN RETURN 1
STRINGLN END
