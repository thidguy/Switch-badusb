REM BOOT DIAGNOSTIC - replicates the bootstrap crash path step by step
REM Run on a FRESH app. Watch the screen - write down EVERY number it prints
REM and which line it dies on. Throwaway run for diagnosis.
DEFAULTDELAY 80
DELAY 6000
STRINGLN PRGEDIT 0:LOADV "TXT:NS2" OUT C$:PRGSET C$
DELAY 3000
STRINGLN RUN
DELAY 8000
STRINGLN ?LEN(ZERO$),LEN(NSR[0]),LEN(NSR[1]),LEN(NSR[2]),LEN(NSR[3])
DELAY 2500
STRINGLN ?NSR[0]+NSR[1]+NSR[2]+NSR[3]
DELAY 2500
STRINGLN REMOVE NSR,&H60000000,&H3FFFFFFF
DELAY 5000
STRINGLN ?LEN(NSR[0]),LEN(NSR[1]),LEN(NSR[2]),LEN(NSR[3])
DELAY 2500
STRINGLN V$=COPY(NSR[0],0,16):B#=U#(LEFT$(V$,4)):C#=U#(MID$(V$,8,4))+16:T#=U#(MID$(V$,12,4))+16:?B#,C#,T#
DELAY 2500
STRINGLN ?PTR(B#,16),PTR(C#-16,16),PTR(T#-16,16)
DELAY 2500
STRINGLN COPY NSR[2],0,P$(T#)+P$(4):COPY NSR[0],0,P$(C#)
DELAY 3000
STRINGLN ?LEN(NSR[0]),NSR[3]
DELAY 2500
STRINGLN ?LEN(KEY(2)),KEY(2)
DELAY 2500
