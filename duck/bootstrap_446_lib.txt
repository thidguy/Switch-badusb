REM ns2jb 4.4.6 primitive library -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2500
STRINGLN ' ns2jb -- 4.4.6 JP demo bootstrap (Switch Lite / FW 23.0.1)
STRINGLN ' Direct-mode runtime order:
STRINGLN '   LOAD "TXT:NS2"
STRINGLN '   RUN
STRINGLN '   REMOVE NSR,&H60000000,&H3FFFFFFF        <- expected: Out of memory
STRINGLN '   V=DISCOVER()
STRINGLN ' NOTE: error branches use the proven v21 idiom only:
STRINGLN '       X$=PLOG(...):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN VAR NSR,V$,D$,S$,A$,X$,OD$,OD2$,ZERO$,CAP$,I,J,R
STRINGLN VAR PHASE=0,B#,C#,T#,E0#,E2#,E3#,KH#,SH#,DGP#,K#,CNT#,PG#,SCAN#,M#,MM#,D#,G#,S2#
STRINGLN ZERO$=CHR$(0)
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000"
STRINGLN COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222"
STRINGLN COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST BOOT READY",4461
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF FAIL(N)
STRINGLN PHASE=-1
STRINGLN PRINT "BOOT FAILED",N
STRINGLN RETURN 0
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10)
STRINGLN PRINT S$
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN IF !PTR(A#,2) THEN RETURN ""
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF UNAIM()
STRINGLN IF OD$!="" THEN COPY NSR[0],16,OD$
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF RD(A#,N)
STRINGLN VAR S$
STRINGLN X$=AIM(A#,N)
STRINGLN S$=COPY(NSR[1],0,N)
STRINGLN UNAIM
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF WW(A$,A#)
STRINGLN VAR N
STRINGLN N=LEN(A$)
STRINGLN IF N<1 THEN RETURN 0
STRINGLN X$=AIM(A#,N)
STRINGLN COPY NSR[1],0,A$
STRINGLN UNAIM
STRINGLN RETURN N
STRINGLN END
STRINGLN COMMON DEF WR#(A#,V#)
STRINGLN RETURN WW(P$(V#),A#)
STRINGLN END
STRINGLN COMMON DEF WRO(A$,A#,O)
STRINGLN VAR N
STRINGLN N=LEN(A$)
STRINGLN IF N<1 THEN RETURN 0
STRINGLN X$=AIM(A#,N+O)
STRINGLN COPY NSR[1],O,A$
STRINGLN UNAIM
STRINGLN RETURN N
STRINGLN END
STRINGLN COMMON DEF S1()
STRINGLN VAR V#
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN IF LEN(V$)!=16 THEN X$=PLOG("S1 SHORT"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4))
STRINGLN E2#=U#(MID$(V$,8,4))
STRINGLN E3#=U#(MID$(V$,12,4))
STRINGLN B#=E0#:C#=E2#+16:T#=E3#+16
STRINGLN IF !PTR(E0#,16) || !PTR(E2#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN IF E2#<=E0# || E3#<=E2# || E3#-E0#>=4096 THEN X$=PLOG("ORDERBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD2$=COPY(NSR[0],48,8)
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("S1 OK "+HX$(E3#-E0#))
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
STRINGLN COMMON DEF S2T()
STRINGLN VAR D$,A$
STRINGLN X$=PLOG("== S2T ==")
STRINGLN A$=RD(E0#,16)
STRINGLN X$=PLOG("A0 "+STR$(A$==V$))
STRINGLN A$=RD(E2#,8)
STRINGLN X$=PLOG("A2 "+STR$(A$==OD2$))
STRINGLN X$=AIM(E3#+16,4)
STRINGLN COPY NSR[1],0,"QQQQ"
STRINGLN X$=PLOG("XW "+STR$(COPY(NSR[0],88,4)=="QQQQ"))
STRINGLN COPY NSR[0],88,"2222"
STRINGLN UNAIM
STRINGLN X$=PLOG("XR "+STR$(COPY(NSR[0],88,4)=="2222")+" "+STR$(NSR[1]=="0000"))
STRINGLN A$=RD(E0#,16)
STRINGLN X$=PLOG("R1 "+STR$(A$==V$))
STRINGLN R=WW("ZZZZ",E3#+16)
STRINGLN X$=PLOG("AIMW "+STR$(R))
STRINGLN X$=PLOG("NZ "+STR$(COPY(NSR[0],88,4)=="ZZZZ"))
STRINGLN COPY NSR[0],88,"2222"
STRINGLN UNAIM
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN IF A$!=V$ THEN RETURN 0
STRINGLN RETURN 1
STRINGLN END
STRINGLN COMMON DEF S3H()
STRINGLN VAR D$
STRINGLN X$=PLOG("== S3H ==")
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN X$=PLOG("KH "+HX$(KH#)+" "+HX$(E0#))
STRINGLN IF !PTR(KH#,16) THEN X$=PLOG("KHBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN R=(KH#-E0#)/2
STRINGLN IF R<0 THEN X$=PLOG("LO NEG"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN D$=COPY(NSR[0],R,8)
STRINGLN DGP#=U#(LEFT$(D$,4))
STRINGLN SH#=U#(MID$(D$,4,4))
STRINGLN X$=PLOG("LDP "+HX$(DGP#)+" "+HX$(SH#))
STRINGLN IF !PTR(DGP#,16) || SH#!=262144 THEN X$=PLOG("ZBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN X$=PLOG("Z5 "+HX$(U#(COPY(NSR[0],R,4))))
STRINGLN X$=AIM(DGP#,2048)
STRINGLN D$=COPY(NSR[1],0,2048)
STRINGLN UNAIM
STRINGLN X$=PLOG("ZERO "+STR$(D$==LEFT$(ZERO$,2048)))
STRINGLN X$=PLOG("PG "+HX$(DGP#)+" LO "+STR$(R))
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN S2#=DGP#
STRINGLN RETURN 1
STRINGLN END
STRINGLN COMMON DEF DISCOVER()
STRINGLN IF PHASE==-1 THEN RETURN FAIL(0)
STRINGLN IF !S1() THEN RETURN 0
STRINGLN IF !S2T() THEN X$=PLOG("S2T FAIL"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN IF !S3H() THEN X$=PLOG("S3H FAIL"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN PHASE=2
STRINGLN PRINT "BOOT OK",4461
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
