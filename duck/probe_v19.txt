REM dprobe_v19 far-region bottom hunt -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1038
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF FLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,T$,J,V#,L#,N,K,E0#,E3#,OD$,KH#,OFF#,DGP#,PG#,DD#,BOT#,V0#,V1#,S$,B$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN S$=COPY(NSR[0],4,4)
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2R ==")
STRINGLN B$="QZ"
STRINGLN FOR I=1 TO 16
STRINGLN B$=B$+B$
STRINGLN NEXT
STRINGLN NSR[1]=B$
STRINGLN KH#=U#(COPY(NSR[0],4,4))
STRINGLN OFF#=(KH#-E0#)/2
STRINGLN D$=COPY(NSR[0],OFF#,8)
STRINGLN DGP#=U#(LEFT$(D$,4))
STRINGLN X$=PLOG("LDP "+HX$(DGP#))
STRINGLN COPY NSR[0],4,S$
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN IF !PTR(DGP#,16) THEN X$=PLOG("DGPBAD"):GOTO @OUT
STRINGLN X$=PLOG("== S3W ==")
STRINGLN BOT#=0:N=0
STRINGLN FOR K=1 TO 128
STRINGLN PG#=DGP#-K*65536
STRINGLN X$=AIM(PG#,16)
STRINGLN T$=COPY(NSR[1],0,16)
STRINGLN COPY NSR[0],16,OD$
STRINGLN V#=U#(T$)
STRINGLN IF V#==0 THEN
STRINGLN IF BOT#==0 THEN BOT#=PG#
STRINGLN N=N+1
STRINGLN IF N<=3 THEN X$=PLOG("Z"+STR$(K)+" "+HX$(PG#))
STRINGLN IF N==1 THEN GOTO @FOUND
STRINGLN ENDIF
STRINGLN IF K MOD 16==0 THEN X$=PLOG("K"+STR$(K)+" "+HX$(V#))
STRINGLN IF K MOD 16==0 THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN @FOUND
STRINGLN IF BOT#>0 THEN
STRINGLN X$=PLOG("BOT "+HX$(BOT#))
STRINGLN X$=AIM(BOT#+65536,2048)
STRINGLN D$=COPY(NSR[1],0,2048)
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("BL "+STR$(LEN(D$)))
STRINGLN IF LEN(D$)==2048 THEN
STRINGLN SAVEV "TXT:BPAGE",D$
STRINGLN N=0
STRINGLN FOR J=0 TO 2040 STEP 4
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN IF V#>=4294967296 && V#<281474976710656 && V#==FLOOR(V#/8)*8 THEN
STRINGLN IF !(V#>=E0#-2097152 && V#<=E0#+2097152) && !(V#>=DGP#-268435456 && V#<=DGP#+268435456) THEN
STRINGLN N=N+1
STRINGLN S$="E"+STR$(J)+" "+HX$(V#)
STRINGLN IF N<=8 THEN X$=PLOG(S$)
STRINGLN ENDIF
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN X$=PLOG("EN "+STR$(N))
STRINGLN ENDIF
STRINGLN ELSE
STRINGLN X$=PLOG("DEEP128")
STRINGLN ENDIF
STRINGLN X$=PLOG("== S4M ==")
STRINGLN FOR K=0 TO 7
STRINGLN DD#=POW(2,K)*1048576
STRINGLN PG#=DGP#-DD#
STRINGLN X$=AIM(PG#,16)
STRINGLN T$=COPY(NSR[1],0,16)
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("M"+STR$(FLOOR(DD#/1048576))+" "+STR$(LEN(T$))+" "+HX$(U#(T$)))
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN @OUT
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
