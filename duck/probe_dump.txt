STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)

STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)

STRINGLN END

STRINGLN END

STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0

STRINGLN D$=COPY(NSR[0],CH*512,512)

STRINGLN ENDIF

STRINGLN NEXT

STRINGLN SAVEV "TXT:PROBEOUT",CAP$
