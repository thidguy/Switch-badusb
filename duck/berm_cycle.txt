REM FULL EXPLOIT CYCLE - BermTheSequel bootstrap on 4.4.6 demo
REM Step 1: Load saved bootstrap into slot 0
REM Step 2: RUN it (prints DIST BOOT READY)
REM Step 3: REMOVE (fires the overflow)
REM Step 4: WAIT for user to type the V=START(...) line from the PC
DEFAULTDELAY 80
DELAY 6000
STRINGLN PRGEDIT 0:LOADV "TXT:NS2" OUT C$:PRGSET C$
DELAY 3000
STRINGLN RUN
DELAY 6000
STRINGLN REMOVE NSR,&H60000000,&H3FFFFFFF
DELAY 4000
PRINT "NOW TYPE THE V=START(...) LINE SHOWN ON THE PC"
